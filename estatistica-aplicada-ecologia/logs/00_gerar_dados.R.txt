
> set.seed(20261026)

> dir.create("dados", showWarnings = FALSE)

> n <- 36

> ambiente <- rep(c("Conservado", "Mosaico", "Restauracao"), 
+     each = 12)

> topografia <- round(runif(n, 3, 30), 1)

> chuva <- round(runif(n, 1800, 3100))

> efeito <- rep(c(0.2, -0.2, 0), each = 12)

> h <- 1.25 + 0.025 * topografia + 2e-04 * chuva + efeito

> h <- round(h + rnorm(n, 0, 0.18), 3)

> riqueza <- rpois(n, lambda = exp(2.4 + 3e-04 * (chuva - 
+     1800)))

> riqueza <- pmax(riqueza, ceiling(exp(h)) + 2)

> aves <- data.frame(local = sprintf("L%02d", 1:n), 
+     ambiente = ambiente, esforco_horas = 4, grupo = "Aves", riqueza = riqueza, 
+     abundancia = riqueza + rpois(n, 55), diversidade_shannon = h, 
+     precipitacao_mm = chuva, topografia_m = topografia, sazonalidade_cv = round(runif(n, 
+         20, 55), 1), cobertura_nativa_pct = round(runif(n, 30, 
+         95), 1))

> write.csv(aves, "dados/comunidades_amazonicas.csv", 
+     row.names = FALSE)

> agua <- data.frame(igarape = sprintf("I%02d", 1:24), 
+     ambiente = rep(c("Conservado", "Mosaico"), each = 12), oxigenio_mg_l = round(c(rnorm(12, 
+         6.6, 0.5), rnorm(12, 5.6, 0.65)), 2), turbidez_ntu = round(rlnorm(24, 
+         log(12), 0.65), 1))

> write.csv(agua, "dados/igarapes.csv", row.names = FALSE)

> antes <- round(rnorm(12, 5.5, 0.6), 2)

> pares <- data.frame(local = sprintf("P%02d", 1:12), 
+     antes_mg_l = antes, depois_mg_l = round(antes + rnorm(12, 
+         0.4, 0.3), 2))

> write.csv(pares, "dados/agua_pareada.csv", row.names = FALSE)

> trat <- rep(c("Baixa", "Media", "Alta"), each = 8)

> perda <- round(rep(c(22, 31, 38), each = 8) + rnorm(24, 
+     0, 3.5), 2)

> dic <- data.frame(recipiente = sprintf("R%02d", 1:24), 
+     tratamento = trat, perda_pct = perda, posicao = sample(1:24))

> write.csv(dic, "dados/experimento_decomposicao.csv",
