> medias <- aggregate(perda_pct ~ tratamento, dados, 
+     mean)

> qm_res <- deviance(modelo)/df.residual(modelo)

> ep <- sqrt(qm_res/8)

> margem <- qt(0.975, df.residual(modelo)) * ep

> medias$inferior <- medias$perda_pct - margem

> medias$superior <- medias$perda_pct + margem

> g <- ggplot(medias, aes(tratamento, perda_pct)) + 
+     geom_errorbar(aes(ymin = inferior, ymax = superior), width = 0.12) + 
+     geom_point(size = 3, color = "#006633") + labs(x = "Umidade", 
+     y = "Perda de massa (%)") + theme_minimal()

> ggsave("figuras/u5_medias.pdf", g, width = 6, height = 3.4)
